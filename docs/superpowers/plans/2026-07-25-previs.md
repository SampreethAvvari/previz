# Previs Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a bidirectional multimodal pre-visualization web app — text→storyboard+locations (autonomous Gemini agent) and image→screenplay (Gemini vision) — deployable to Cloud Run + Firebase.

**Architecture:** FastAPI backend on Cloud Run hosts a bounded Gemini function-calling agent (Mode A) that plans shots and calls `generate_storyboard_image` + `find_locations` tools, streaming events over SSE. A separate `/analyze-image` endpoint (Mode B) does a one-shot Gemini vision call → screenplay. React+Vite+Tailwind frontend on Firebase Hosting consumes the SSE stream and renders a storyboard grid + location cards, or the screenplay.

**Tech Stack:** Python 3.11 · FastAPI · `google-genai` SDK (Gemini 2.5 + Imagen) · Google Maps Places API (New) · React + Vite + TypeScript + Tailwind · Firebase Hosting · Cloud Run.

## Global Constraints

- **Python 3.11+**, **Node 20+**.
- **Google Gen AI SDK:** package `google-genai`, imported `from google import genai`. **Its method signatures change between releases — before implementing any `gemini_client.py` / image / vision call, verify the current API via the context7 MCP (`resolve-library-id` → `query-docs` for "google-genai") or official docs. Do not trust a signature from memory.**
- **Bounded agent:** max **6** shots per run, max **15** total tool calls. Enforce in code — never rely on the model to self-limit.
- **No secrets in code.** Load from env: `.env` locally, Secret Manager in prod. Keys: `GEMINI_API_KEY`, `GOOGLE_MAPS_API_KEY`.
- **Images returned as base64 `data:` URLs** — no blob storage for the demo.
- **Every Google API call:** wrap in retry-once + graceful fallback (never crash the request).
- **Keep files focused** — one responsibility each; split when a file grows past its purpose.
- **Commits** include the trailer: `Co-Authored-By: Claude Opus 4.8 (1M context) <noreply@anthropic.com>`.

import httpx
from app.config import settings
from app.models import LocationSuggestion

_URL = "https://places.googleapis.com/v1/places:searchText"

def find_locations(scene_description: str, region: str | None = None) -> list[LocationSuggestion]:
    query = scene_description if not region else f"{scene_description} in {region}"
    headers = {
        "X-Goog-Api-Key": settings.google_maps_api_key,
        "X-Goog-FieldMask": "places.displayName,places.formattedAddress,places.location,places.id",
    }
    try:
        r = httpx.post(_URL, headers=headers, json={"textQuery": query, "maxResultCount": 3}, timeout=10)
        r.raise_for_status()
        places = r.json().get("places", [])
    except Exception:
        return []
    out = []
    for p in places[:3]:
        loc = p.get("location", {})
        pid = p.get("id", "")
        out.append(LocationSuggestion(
            name=p.get("displayName", {}).get("text", "Unknown"),
            address=p.get("formattedAddress", ""),
            lat=loc.get("latitude", 0.0), lng=loc.get("longitude", 0.0),
            maps_url=f"https://www.google.com/maps/place/?q=place_id:{pid}",
        ))
    return out
```

import asyncio
from typing import AsyncIterator
from google.genai import types
from app.config import settings
from app.gemini_client import get_client
from app.models import StylePreset, Shot, ShotPlanned, ImageReady, LocationFound, Done, ErrorEvent
from app.tools.storyboard import generate_storyboard_image
from app.tools.locations import find_locations

def plan_shots(scene: str, style: StylePreset) -> list[Shot]:
    resp = get_client().models.generate_content(
        model="gemini-2.5-flash",
        contents=(f"Break this film scene into at most {settings.max_shots} storyboard shots. "
                  f"Return JSON list of objects with 'index' and 'description'. Scene:\n{scene}"),
        config=types.GenerateContentConfig(response_mime_type="application/json"),
    )
    import json
    data = json.loads(resp.text)[: settings.max_shots]
    return [Shot(index=i, description=d.get("description", "")) for i, d in enumerate(data)]

async def run_agent(scene: str, style: StylePreset) -> AsyncIterator[object]:
    try:
        shots = await asyncio.to_thread(plan_shots, scene, style)
    except Exception:
        yield ErrorEvent(message="Failed to plan shots"); yield Done(); return
    yield ShotPlanned(shots=shots)
    calls = 0
    for shot in shots:
        if calls >= settings.max_tool_calls: break
        img = await asyncio.to_thread(generate_storyboard_image, shot.description, style); calls += 1
        yield ImageReady(shot_index=shot.index, image_data_url=img)
    if calls < settings.max_tool_calls:
        locs = await asyncio.to_thread(find_locations, scene, None); calls += 1
        yield LocationFound(scene=scene, locations=locs)
    yield Done()
```

from fastapi import FastAPI, UploadFile, File, Form
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import StreamingResponse
from pydantic import BaseModel
from app.models import StylePreset
from app.agent import run_agent
from app.screenplay import analyze_image_to_screenplay
from app.sse import sse

app = FastAPI(title="Previs")
app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_methods=["*"], allow_headers=["*"])

class GenerateReq(BaseModel):
    scene: str
    style: StylePreset = StylePreset()

@app.get("/healthz")
def healthz(): return {"ok": True}

@app.post("/generate")
async def generate(req: GenerateReq):
    async def stream():
        async for ev in run_agent(req.scene, req.style):
            yield sse(ev)
    return StreamingResponse(stream(), media_type="text/event-stream")

@app.post("/analyze-image")
async def analyze(file: UploadFile = File(...), style: str | None = Form(None)):
    preset = StylePreset(**json.loads(style)) if style else None
    data = await file.read()
    return {"screenplay": analyze_image_to_screenplay(data, file.content_type or "image/png", preset)}

import type { PrevisEvent, StylePreset } from "./types";
const BASE = import.meta.env.VITE_API_BASE;

export async function streamGenerate(scene: string, style: StylePreset, onEvent: (e: PrevisEvent) => void) {
  const res = await fetch(`${BASE}/generate`, {
    method: "POST", headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ scene, style }),
  });
  const reader = res.body!.getReader(); const dec = new TextDecoder(); let buf = "";
  for (;;) {
    const { value, done } = await reader.read(); if (done) break;
    buf += dec.decode(value, { stream: true });
    const parts = buf.split("\n\n"); buf = parts.pop() || "";
    for (const p of parts) if (p.startsWith("data: ")) onEvent(JSON.parse(p.slice(6)));
  }
}

export async function analyzeImage(file: File, style?: StylePreset): Promise<string> {
  const fd = new FormData(); fd.append("file", file);
  if (style) fd.append("style", JSON.stringify(style));
  const res = await fetch(`${BASE}/analyze-image`, { method: "POST", body: fd });
  return (await res.json()).screenplay;
}
